cmake_minimum_required(VERSION 3.16)
project(FDCut VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(FD_FETCH_HPF "Download checksum-pinned HPF (separate research license)" ON)
set(FD_HPF_SOURCE "" CACHE FILEPATH "Existing official HPF 3.23 pseudo.c")
if(NOT FD_HPF_SOURCE)
  if(NOT FD_FETCH_HPF)
    message(FATAL_ERROR "Provide FD_HPF_SOURCE or enable FD_FETCH_HPF")
  endif()
  include(FetchContent)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  FetchContent_Declare(hpf
    URL https://riot.ieor.berkeley.edu/Applications/Pseudoflow/pseudo-max-3.23_2.tar
    URL_HASH SHA256=722bf7d8f8ad7dedb776171e71742e8f5643d1f8cb266b900e2147ff2b390903)
  FetchContent_GetProperties(hpf)
  if(NOT hpf_POPULATED)
    FetchContent_Populate(hpf)
  endif()
  file(GLOB_RECURSE HPF_CANDIDATES "${hpf_SOURCE_DIR}/pseudo.c")
  list(LENGTH HPF_CANDIDATES HPF_COUNT)
  if(NOT HPF_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected exactly one HPF pseudo.c")
  endif()
  list(GET HPF_CANDIDATES 0 FD_HPF_SOURCE)
endif()
get_filename_component(FD_HPF_SOURCE "${FD_HPF_SOURCE}" ABSOLUTE)
add_library(hpf_adapter STATIC src/hpf_memory.c)
set_property(TARGET hpf_adapter PROPERTY C_STANDARD 90)
set_property(TARGET hpf_adapter PROPERTY C_EXTENSIONS ON)
target_compile_definitions(hpf_adapter PRIVATE FIFO_BUCKET FD_HPF_SOURCE="${FD_HPF_SOURCE}")
add_library(frontdoor STATIC src/enumerate.cpp src/frontdoor.cpp
  src/kernels/array_compile.cpp src/kernels/sweep_csr.cpp src/kernels/component_sweep.cpp)
target_include_directories(frontdoor PUBLIC include)
target_link_libraries(frontdoor PRIVATE hpf_adapter)
target_compile_options(frontdoor PRIVATE -Wall -Wextra)
find_package(Threads REQUIRED)
add_executable(fdtool src/main.cpp)
target_link_libraries(fdtool PRIVATE frontdoor Threads::Threads)
add_executable(fdsolveir tools/solve_ir.cpp)
target_link_libraries(fdsolveir PRIVATE frontdoor Threads::Threads)
foreach(pair "fdratioe2;generate_ratio_e2" "fde3gen;generate_witness_e3" "fdfirsthop;generate_firsthop_e2")
  list(GET pair 0 target)
  list(GET pair 1 source)
  add_executable(${target} tools/${source}.cpp)
  target_link_libraries(${target} PRIVATE frontdoor Threads::Threads)
endforeach()
foreach(target pack_graph generate_large_dag graph_convert)
  add_executable(${target} tools/${target}.cpp)
  target_include_directories(${target} PRIVATE include)
endforeach()
add_executable(read_graph examples/read_graph.cpp)
target_include_directories(read_graph PRIVATE include)
enable_testing()
add_test(NAME native_selftest COMMAND fdtool selftest)
add_executable(query_common_selftest tests/query_common.cpp)
target_include_directories(query_common_selftest PRIVATE include)
add_test(NAME query_common_selftest COMMAND query_common_selftest)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME method_names COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/method_names.py $<TARGET_FILE:fdtool> $<TARGET_FILE:fdsolveir>)
  add_test(NAME release_boundaries COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/release_boundaries.py)
  foreach(test enumeration_limits oracle input_capacity benchmark12_workloads)
    add_test(NAME ${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/${test}.py $<TARGET_FILE:fdtool>)
  endforeach()
  foreach(test solve_ir solve_ir_compact)
    add_test(NAME ${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/${test}.py $<TARGET_FILE:fdsolveir>)
  endforeach()
  add_test(NAME graph_formats COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/graph_formats.py $<TARGET_FILE:graph_convert>)
  add_test(NAME large_graph_io COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/large_graph_io.py $<TARGET_FILE:pack_graph>)
  add_test(NAME large_dag_generator COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/large_dag_generator.py $<TARGET_FILE:generate_large_dag>)
  add_test(NAME ratio_e2 COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/ratio_e2.py $<TARGET_FILE:fdratioe2>)
  add_test(NAME witness_e3 COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/witness_e3.py $<TARGET_FILE:fde3gen>)
  add_test(NAME query_to_spec COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/query_to_spec.py $<TARGET_FILE:fdsolveir>)
endif()
install(TARGETS fdtool fdsolveir fdratioe2 fde3gen fdfirsthop graph_convert RUNTIME DESTINATION bin)
